6. C — Factorial

#include <stdio.h>

int main(void) {
    int n;
    unsigned long long fact = 1;

    printf("Enter a number (0-20): ");

    if (scanf("%d", &n) != 1 ||
        n < 0 || n > 20) {
        printf("Invalid input\n");
        return 1;
    }

    for (int i = 2; i <= n; i++)
        fact *= i;

    printf("Factorial = %llu\n", fact);

    return 0;
}

Linear Search

Checks array elements one by one until the target is found.

int a[] = {5, 10, 15, 20};
int target = 15;
int found = -1;

for (int i = 0; i < 4; i++) {
    if (a[i] == target) {
        found = i;
        break;
    }
}

printf("%d", found);

Bubble Sort

Repeatedly compares neighbouring values and swaps them when needed.

int a[] = {5, 2, 4, 1};
int n = 4;

for (int i = 0; i < n - 1; i++) {
    for (int j = 0; j < n - 1 - i; j++) {
        if (a[j] > a[j + 1]) {
            int temp = a[j];
            a[j] = a[j + 1];
            a[j + 1] = temp;
        }
    }
}
